// 60 pieces across 3 depth tiers (far = small/slow/dim, near = big/fast).
// Per piece: GSAP runs three tweens for the play window (~3s):
gsap.set(el, { scale, transformPerspective: 600, autoAlpha: 0 });

// 1. Fall: -10vh to 110vh, fading in at 8% and out at 90%.
gsap.to(el, { keyframes: {
  '0%':   { y: '-10vh', autoAlpha: 0 },
  '8%':   { autoAlpha: opacity },
  '90%':  { autoAlpha: opacity },
  '100%': { y: '110vh', autoAlpha: 0 }, easeEach: 'none',
}, duration, delay, repeat: plays - 1, ease: 'none' });

// 2. Spin + 3D tumble, synced to the fall.
gsap.to(el, { rotation, rotationX, rotationY, duration, delay, repeat: plays - 1, ease: 'none' });

// 3. Side-to-side sway, continuous and unsynced for variety.
gsap.to(el, { x: swayAmp, duration: swayDur, ease: 'sine.inOut', yoyo: true, repeat: -1 });
